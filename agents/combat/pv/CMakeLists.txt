# Optional ONNX-backed PV agent. Prepare with apps/pv/setup_runtime.sh.
set(PV_ORT_ROOT "${PROJECT_SOURCE_DIR}/build/deps/onnxruntime" CACHE PATH "ONNX Runtime C++ headers/library")
if(NOT EXISTS "${PV_ORT_ROOT}/include/onnxruntime_cxx_api.h" OR NOT EXISTS "${PV_ORT_ROOT}/lib/libonnxruntime.so.1")
    message(STATUS "PV not built: run apps/pv/setup_runtime.sh for ONNX Runtime C++ support")
    return()
endif()
add_library(pv_onnx STATIC features.cpp evaluator.cpp search.cpp turn_search.cpp turn_pimc.cpp turn_targets.cpp real_turn.cpp)
target_include_directories(pv_onnx SYSTEM PRIVATE "${PV_ORT_ROOT}/include")
target_link_libraries(pv_onnx PUBLIC combat_core "${PV_ORT_ROOT}/lib/libonnxruntime.so.1")
target_link_options(pv_onnx PUBLIC "-Wl,-rpath,${PV_ORT_ROOT}/lib")
target_compile_options(pv_onnx PRIVATE -Wall -Wextra -Wpedantic)
if(NOT STSRL_ARROW_LIB OR NOT STSRL_PARQUET_LIB)
    message(STATUS "PV not built: pyarrow wheel (Arrow / Parquet) not found in ${STSRL_PYTHON}")
    return()
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(pv_worker "${PROJECT_SOURCE_DIR}/apps/pv/worker.cpp")
target_include_directories(pv_worker SYSTEM PRIVATE "${STSRL_PYARROW_DIR}/include")
target_include_directories(pv_worker PRIVATE "${PROJECT_SOURCE_DIR}")
target_link_libraries(pv_worker PRIVATE pv_onnx ${STSRL_PARQUET_LIB} ${STSRL_ARROW_LIB} OpenSSL::Crypto)
target_link_options(pv_worker PRIVATE "-Wl,-rpath,${STSRL_PYARROW_DIR}")
target_compile_options(pv_worker PRIVATE -Wall -Wextra -Wpedantic)
